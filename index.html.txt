        function extraerTextoCelda(json, colIndexPref, rowIndexPref) {
            if (!json || !json.table) return '';
            const rows = json.table.rows || [];
            
            for (let r = 0; r < Math.min(rows.length, 6); r++) {
                if (rows[r]?.c && rows[r].c[colIndexPref] && rows[r].c[colIndexPref].v != null) {
                    const v = String(rows[r].c[colIndexPref].v).trim();
                    if (v !== '' && v !== 'null') return v;
                }
            }
            for (let r = 0; r < Math.min(rows.length, 6); r++) {
                if (rows[r]?.c && rows[r].c[4] && rows[r].c[4].v != null) {
                    const v = String(rows[r].c[4].v).trim();
                    if (v !== '' && v !== 'null') return v;
                }
            }
            if (json.table.cols) {
                if (json.table.cols[colIndexPref]?.label && String(json.table.cols[colIndexPref].label).trim() !== '') {
                    return String(json.table.cols[colIndexPref].label).trim();
                }
                if (json.table.cols[4]?.label && String(json.table.cols[4].label).trim() !== '') {
                    return String(json.table.cols[4].label).trim();
                }
            }
            for (let r = 0; r < rows.length; r++) {
                if (rows[r]?.c) {

        Promise.all([
            fetch(urlPeliculas).then(r => r.text()),
            fetch(urlSeries).then(r => r.text()),
            fetch(urlSagas).then(r => r.text()),
            fetch(urlPacks).then(r => r.text()),
            fetch(urlPaseVip).then(r => r.text())
        ]).then(([dataPeli, dataSerie, dataSaga, dataPack, dataVip]) => {
            // 1. PELÍCULAS
            const jsonPeli = JSON.parse(dataPeli.substring(47).slice(0, -2));
            const rowsPelis = jsonPeli.table.rows || [];
            
            infoGeneralPeliculas = extraerTextoCelda(jsonPeli, 5, 0);

            todasLasPeliculas = rowsPelis.map(row => ({
                tipo: 'pelicula',
                nombre: row.c[0] ? String(row.c[0].v).trim() : '',
                año: row.c[1] ? String(row.c[1].v).trim() : '',
                generos: row.c[2] ? String(row.c[2].v).trim() : '',
                imagen: row.c[3] ? String(row.c[3].v).trim() : '',
                posicionCarousel: (row.c[6] && row.c[6].v != null && !isNaN(parseInt(row.c[6].v))) ? parseInt(row.c[6].v) : null
            })).filter(peli => peli.nombre !== 'NOMBRE' && peli.nombre !== '');

            // 2. SERIES
            const jsonSerie = JSON.parse(dataSerie.substring(47).slice(0, -2));
            const rowsSeries = jsonSerie.table.rows || [];

            infoGeneralSeries = extraerTextoCelda(jsonSerie, 5, 0);

            todasLasSeries = rowsSeries.map(row => ({
                tipo: 'serie',
                nombre: row.c[0] ? String(row.c[0].v).trim() : '',
                año: row.c[1] ? String(row.c[1].v).trim() : '',
                episodios: (row.c[2] && row.c[2].v != null) ? String(row.c[2].v).trim() : '',
                pack: (row.c[3] && row.c[3].v != null) ? String(row.c[3].v).trim() : '',
                imagen: row.c[4] ? String(row.c[4].v).trim() : ''
            })).filter(serie => serie.nombre !== 'NOMBRE' && serie.nombre !== '');

            // 3. SAGAS
            const jsonSaga = JSON.parse(dataSaga.substring(47).slice(0, -2));
            const rowsSagas = jsonSaga.table.rows;
            const filaPrecios = rowsSagas[0]?.c || [];
            const filaCantidades = rowsSagas[1]?.c || [];
            todasLasSagas = [];

            for (let i = 2; i < rowsSagas.length; i++) {
                const celdas = rowsSagas[i]?.c;
                if (!celdas || !celdas[0] || celdas[0].v === null || String(celdas[0].v).trim() === '') continue;

                const nombreSaga = String(celdas[0].v).trim();
                let ultimaColumna = 1;
                let imagenes = [];

                for (let j = 1; j < celdas.length; j++) {
                    if (celdas[j] && celdas[j].v !== null && String(celdas[j].v).trim() !== '') {
                        imagenes.push(String(celdas[j].v).trim());
                        ultimaColumna = j;
                    }
                }

                if (imagenes.length === 0) continue;

                let precio = (filaPrecios[ultimaColumna] && filaPrecios[ultimaColumna].v !== null)
                    ? String(filaPrecios[ultimaColumna].v).trim()
                    : '0';
                if (!precio.startsWith('$')) precio = '$' + precio;

                const cantidadX = (filaCantidades[ultimaColumna] && filaCantidades[ultimaColumna].v !== null)
                    ? String(filaCantidades[ultimaColumna].v).trim()
                    : String(imagenes.length);

                todasLasSagas.push({
                    tipo: 'saga',
                    nombre: nombreSaga,
                    precio: precio,
                    cantidad: cantidadX,
                    imagenesSlider: [...imagenes]
                });
            }

            // 4. PACKS DIGITALES
            const jsonPack = JSON.parse(dataPack.substring(47).slice(0, -2));
            const rowsPacks = jsonPack.table.rows || [];
            
            infoGeneralPacks = extraerTextoCelda(jsonPack, 4, 0);

            todosLosPacks = rowsPacks.map(row => {
                let precio = row.c[1] ? String(row.c[1].v).trim() : '';
                if (precio && !precio.startsWith('$')) precio = '$' + precio;

                return {
                    tipo: 'pack',
                    nombre: row.c[0] ? String(row.c[0].v).trim() : '',
                    precio: precio,
                    contenido: (row.c[2] && row.c[2].v != null) ? String(row.c[2].v).trim() : '',
                    imagen: (row.c[3] && row.c[3].v != null) ? String(row.c[3].v).trim() : ''
                };
            }).filter(pack => pack.nombre !== '' && pack.nombre !== 'NOMBRE');

            // 5. PASE VIP (Fila 1 Películas / Fila 2 Series)
            const jsonVip = JSON.parse(dataVip.substring(47).slice(0, -2));
            const rowsVip = jsonVip.table.rows || [];
            const dataRowsVip = rowsVip.filter(r => r.c && r.c[0] && r.c[0].v && String(r.c[0].v).trim().toUpperCase() !== 'NOMBRE');

            function parsearFilaVip(r) {
                if (!r || !r.c) return null;
                let precio = r.c[1] ? String(r.c[1].v).trim() : '';
                if (precio && !precio.startsWith('$')) precio = '$' + precio;
                return {
                    nombre: r.c[0] ? String(r.c[0].v).trim() : '',
                    precio: precio,
                    imagen: r.c[2] ? String(r.c[2].v).trim() : '',
                    textoA: r.c[3] ? String(r.c[3].v).trim() : '',
                    textoB: r.c[4] ? String(r.c[4].v).trim() : ''
                };
            }

            if (dataRowsVip.length > 0) datosPaseVip.peliculas = parsearFilaVip(dataRowsVip[0]);
            if (dataRowsVip.length > 1) datosPaseVip.series = parsearFilaVip(dataRowsVip[1]);

            actualizarInterfazCarrito();
            verificarRutaDirecta();
        }).catch(err => console.error('Error cargando bases de datos:', err));

        function obtener15EstrenosCarousel() {
            const pelisAnioActual = todasLasPeliculas.filter(p => String(p.año).trim() === String(ANIO_ACTUAL));
            let slots = Array(15).fill(null);

            pelisAnioActual.forEach(p => {
                if (p.posicionCarousel !== null && p.posicionCarousel >= 1 && p.posicionCarousel <= 15) {
                    if (!slots[p.posicionCarousel - 1]) {
                        slots[p.posicionCarousel - 1] = p;
                    }
                }
            });

            let indexPeli = 0;
            for (let i = 0; i < 15; i++) {
                if (!slots[i]) {
                    while (indexPeli < pelisAnioActual.length && slots.includes(pelisAnioActual[indexPeli])) {
                        indexPeli++;
                    }
                    if (indexPeli < pelisAnioActual.length) {
                        slots[i] = pelisAnioActual[indexPeli];
                        indexPeli++;
                    }
                }
            }

            return slots.filter(p => p !== null);
        }

        function cambiarTitulo(titulo, modo = 'normal') {
            const titleContainer = document.getElementById('section-title-container');
            const titleText = document.getElementById('section-title-text');
            const sortWrapper = document.getElementById('sort-wrapper');
            const btnVerLista = document.getElementById('ver-lista-btn');
            const btnToggleView = document.getElementById('toggle-view-btn');
            const btnInfo = document.getElementById('btn-info-section');

            titleContainer.classList.remove('search-active', 'modo-series', 'modo-sagas', 'modo-packs');

            sortWrapper.style.display = 'block';
            btnVerLista.style.display = 'none';
            btnToggleView.style.display = 'none';
            btnInfo.style.display = 'none';

            if (modo === 'busqueda') {
                titleContainer.classList.add('search-active');
                titleText.innerHTML = `
                    <span style="display: inline-flex; align-items: center; gap: 8px;">
                        ${iconoLupaSVG}
                        <span>${titulo.toUpperCase()}</span>
                    </span>
                `;
                btnInfo.style.display = 'inline-flex';
                btnInfo.innerHTML = (tipoBusquedaActual === 'series') ? `${iconoSerieSVG15} INFO` : `${iconoPeliculaSVG15} INFO`;
            } else if (modo === 'series') {
                titleContainer.classList.add('modo-series');
                titleText.innerHTML = `
                    <span style="display: inline-flex; align-items: center; gap: 8px;">
                        ${iconoSerieSVG15}
                        <span>${titulo.toUpperCase()}</span>
                    </span>
                `;
                btnInfo.style.display = 'inline-flex';
                btnInfo.innerHTML = `${iconoSerieSVG15} INFO`;
            } else if (modo === 'sagas') {
                titleContainer.classList.add('search-active', 'modo-sagas');
                titleText.innerHTML = `
                    <span style="display: inline-flex; align-items: center; gap: 8px;">
                        ${iconoClaquetaSVG}
                        <span>${titulo.toUpperCase()}</span>
                    </span>
                `;
                sortWrapper.style.display = 'none';
                btnVerLista.style.display = 'inline-flex';
            } else if (modo === 'packs') {
                titleContainer.classList.add('search-active', 'modo-packs');
                titleText.innerHTML = `
                    <span style="display: inline-flex; align-items: center; gap: 8px;">
                        ${iconoPacksCarpetaSVG}
                        <span>${titulo.toUpperCase()}</span>
                    </span>
                `;
                sortWrapper.style.display = 'none';
                btnVerLista.style.display = 'inline-flex';
                btnToggleView.style.display = 'inline-flex';
                actualizarBotonVistaPacks();
            } else {
                titleText.textContent = titulo.toUpperCase();
                btnInfo.style.display = 'inline-flex';
                btnInfo.innerHTML = `${iconoPeliculaSVG15} INFO`;
            }
        }

        function verificarRutaDirecta() {
            const hash = window.location.hash.toLowerCase();
            
            // Enlace directo a un pack digital: #pack=Nombre o #pack-Nombre
            if (hash.startsWith('#pack=') || hash.startsWith('#pack-')) {
                const nombrePack = decodeURIComponent(hash.replace(/^#pack[=-]/, '')).trim();
                mostrarPackDirecto(nombrePack);
                return;
            }

            if (hash === '#estrenos') {
                mostrarEstrenos();
            } else if (hash === '#recien-anadidas' || hash === '#mas-recientes') {
                mostrarMasRecientes();
            } else if (hash === '#sagas' || hash === '#promo-sagas') {
                mostrarSagas();
            } else if (hash === '#pack-digital' || hash === '#packs') {
                mostrarPacks();
            } else if (hash === '#series') {
                mostrarSeries();
            } else if (hash === '#pase-vip' || hash === '#pase-vip-peliculas') {
                mostrarPaseVip('peliculas');
            } else if (hash === '#pase-vip-series') {
                mostrarPaseVip('series');
            } else if (hash && hash !== '#inicio' && hash !== '#') {
                const genero = decodeURIComponent(hash.substring(1));
                filtrarPorGenero(genero);
            } else {
                mostrarInicio();
            }
        }

        function ordenarPeliculas(tipo) {
            if (esSeccionEstrenos) {
                elementosActivos.sort((a, b) => {
                    const anioA = Number(a.año) || 0;
                    const anioB = Number(b.año) || 0;
                    if (anioA !== anioB) return anioB - anioA;

                    const nomA = String(a.nombre || '').trim();
                    const nomB = String(b.nombre || '').trim();

                    if (tipo === 'recientes') {
                        return todasLasPeliculas.indexOf(a) - todasLasPeliculas.indexOf(b);
                    }
                    if (tipo === 'antiguas') {
                        return todasLasPeliculas.indexOf(b) - todasLasPeliculas.indexOf(a);
                    }
                    if (tipo === 'az' || tipo === 'nombre') {
                        return nomA.localeCompare(nomB, 'es', { sensitivity: 'base', numeric: true });
                    }
                    if (tipo === 'za') {
                        return nomB.localeCompare(nomA, 'es', { sensitivity: 'base', numeric: true });
                    }
                    return 0;
                });
            } else {
                switch (tipo) {
                    case 'recientes': {
                        const listaBase = modoActual === 'series' ? todasLasSeries : todasLasPeliculas;
                        elementosActivos.sort((a, b) => listaBase.indexOf(a) - listaBase.indexOf(b));
                        break;
                    }
                    case 'anio':
                        elementosActivos.sort((a, b) => Number(b.año) - Number(a.año));
                        break;
                    case 'az':
                    case 'nombre':
                        elementosActivos.sort((a, b) => 
                            String(a.nombre || '').trim().localeCompare(String(b.nombre || '').trim(), 'es', { sensitivity: 'base', numeric: true })
                        );
                        break;
                    case 'za':
                        elementosActivos.sort((a, b) => 
                            String(b.nombre || '').trim().localeCompare(String(a.nombre || '').trim(), 'es', { sensitivity: 'base', numeric: true })
                        );
                        break;
                    case 'antiguas':
                        elementosActivos.sort((a, b) => Number(a.año) - Number(b.año));
                        break;
                }
            }
            sortDropdown.classList.remove('open');
            mostrarContenido(true);
        }

        // ================= DESLIZAMIENTO TÁCTIL =================
        let touchStartX = 0;
        let touchCurrentX = 0;
        let isDraggingModal = false;
        const modalContainer = document.querySelector('.modal-img-container');

        if (modalContainer) {
            modalContainer.addEventListener('touchstart', function(e) {
                if (modalSagaImagenes.length <= 1) return;
                touchStartX = e.touches[0].clientX;
                touchCurrentX = touchStartX;
                isDraggingModal = true;
                const track = document.getElementById('modal-slider-track');
                track.style.transition = 'none';
            }, { passive: true });

            modalContainer.addEventListener('touchmove', function(e) {
                if (!isDraggingModal || modalSagaImagenes.length <= 1) return;
                touchCurrentX = e.touches[0].clientX;
                let diffX = touchCurrentX - touchStartX;

                if (modalSagaIndexActual === 0 && diffX > 0) diffX = 0;
                else if (modalSagaIndexActual === modalSagaImagenes.length - 1 && diffX < 0) diffX = 0;

                const containerWidth = modalContainer.offsetWidth || 300;
                const dragPercent = (diffX / containerWidth) * 100;
                const currentBasePercent = -modalSagaIndexActual * 100;

                const track = document.getElementById('modal-slider-track');
                track.style.transform = `translateX(${currentBasePercent + dragPercent}%)`;
            }, { passive: true });

            modalContainer.addEventListener('touchend', function(e) {
                if (!isDraggingModal || modalSagaImagenes.length <= 1) return;
                isDraggingModal = false;
                
                let diffX = touchCurrentX - touchStartX;
                if (modalSagaIndexActual === 0 && diffX > 0) diffX = 0;
                if (modalSagaIndexActual === modalSagaImagenes.length - 1 && diffX < 0) diffX = 0;

                const containerWidth = modalContainer.offsetWidth || 300;
                const threshold = Math.min(containerWidth * 0.2, 50);

                const track = document.getElementById('modal-slider-track');
                track.style.transition = 'transform 0.35s cubic-bezier(0.25, 1, 0.5, 1)';

                if (diffX < -threshold && modalSagaIndexActual < modalSagaImagenes.length - 1) {
                    modalSagaIndexActual++;
                } else if (diffX > threshold && modalSagaIndexActual > 0) {
                    modalSagaIndexActual--;
                }

                track.style.transform = `translateX(-${modalSagaIndexActual * 100}%)`;
                actualizarBotonesModalSaga();
            }, { passive: true });
        }

        const searchModal = document.getElementById('search-modal');
        const searchInput = document.getElementById('search-input');
        const clearSearchBtn = document.getElementById('clear-search-btn');
        const searchResultsDropdown = document.getElementById('search-results-dropdown');
        const viewAllResultsBtn = document.getElementById('view-all-results-btn');
        const btnBuscarPeliculas = document.getElementById('btn-buscar-peliculas');
        const btnBuscarSeries = document.getElementById('btn-buscar-series');
        let tipoBusquedaActual = 'peliculas';

        function normalizarTexto(texto) {
            return String(texto || '')
                .normalize('NFD')
                .replace(/[\u0300-\u036f]/g, '')
                .toLowerCase();
        }

        function abrirBuscador() {
            registrarEstadoModal();
            bloquearScroll(true);
            searchModal.style.display = 'flex';
            setTimeout(() => searchInput.focus(), 50);
        }

        function cerrarBuscador(limpiarHistorial = true) {
            if (limpiarHistorial) limpiarEstadoModal();
            bloquearScroll(false);
            searchModal.style.display = 'none';
        }

        function limpiarTextoBusqueda() {
            searchInput.value = '';
            clearSearchBtn.style.display = 'none';
            searchResultsDropdown.style.display = 'none';
            viewAllResultsBtn.style.display = 'none';
            searchInput.focus();
        }

        function cambiarTipoBusqueda(tipo) {
            if (tipo === tipoBusquedaActual) return;
            tipoBusquedaActual = tipo;
            btnBuscarPeliculas.classList.toggle('active', tipo === 'peliculas');
            btnBuscarSeries.classList.toggle('active', tipo === 'series');
            ejecutarBusqueda(searchInput.value.trim());
        }

        function ejecutarBusqueda(query) {
            terminoBusquedaActual = query;

            if (query.length > 0) {
                clearSearchBtn.style.display = 'flex';

                const listaBase = tipoBusquedaActual === 'series' ? todasLasSeries : todasLasPeliculas;
                const queryNormalizada = normalizarTexto(query);
                const resultados = listaBase.filter(item => 
                    normalizarTexto(item.nombre).includes(queryNormalizada)
                );

                if (resultados.length > 0) {
                    searchResultsDropdown.innerHTML = '';
                    const maxResultados = resultados.slice(0, 4);

                    maxResultados.forEach(item => {
                        const el = document.createElement('div');
                        el.className = 'search-result-item';
                        el.onclick = () => {
                            abrirModalPeliculaOSerie(item);
                        };

                        const iconoTipo = item.tipo === 'serie' ? iconoSerieSVG15 : iconoPeliculaSVG15;
                        const etiquetaTipo = item.tipo === 'serie' ? 'SERIE' : 'PELÍCULA';

                        el.innerHTML = `
                            <img src="${item.imagen}" alt="${item.nombre}" class="search-result-thumb">
                            <div class="search-result-info">
                                <span class="search-result-title">${item.nombre}</span>
                                <span class="search-result-meta">
                                    ${item.año} 
                                    ${iconoTipo}
                                    ${etiquetaTipo}
                                </span>
                            </div>
                        `;
                        searchResultsDropdown.appendChild(el);
                    });

                    searchResultsDropdown.style.display = 'flex';
                    viewAllResultsBtn.style.display = 'block';
                } else {
                    searchResultsDropdown.innerHTML = '<div style="color: #666; text-align: center; padding: 10px; font-size: 13px;">No se encontraron resultados</div>';
                    searchResultsDropdown.style.display = 'flex';
                    viewAllResultsBtn.style.display = 'none';
                }
            } else {
                clearSearchBtn.style.display = 'none';
                searchResultsDropdown.style.display = 'none';
                viewAllResultsBtn.style.display = 'none';
            }
        }

        searchInput.addEventListener('input', function() {
            ejecutarBusqueda(searchInput.value.trim());
        });

        function verTodosLosResultados() {
            if (!terminoBusquedaActual) return;

            document.body.classList.remove('modo-vip-bg');
            modoActual = tipoBusquedaActual;
            const listaBase = tipoBusquedaActual === 'series' ? todasLasSeries : todasLasPeliculas;
            const queryNormalizada = normalizarTexto(terminoBusquedaActual);

            elementosActivos = listaBase.filter(item => 
                normalizarTexto(item.nombre).includes(queryNormalizada)
            );

            cambiarTitulo(terminoBusquedaActual, 'busqueda');
            cerrarBuscador();
            mostrarContenido(true); 
        }

        searchModal.addEventListener('click', function(e) {
            if (e.target === searchModal) cerrarBuscador();
        });
